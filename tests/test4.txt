VAR_GLOBAL {
    INTEGER T,J,I;
}
DECLARATION {}
INSTRUCTION {
    %% Initialize result
    T = 0;

    %% Perform operations directly
    FOR(I = 1 : 1 : 3) {
        FOR(J = 1 : 1 : 3) {
            T = T + (-5 * 4) + (4 * -3);  %% Direct operation with signed and unsigned numbers
        }
    }

    %% Write the total
    WRITE("Total: ", T);
}
