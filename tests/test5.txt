VAR_GLOBAL {
    INTEGER S,I,J;
}
DECLARATION {}
INSTRUCTION {
    %% Initialize sum
    S = 0;

    %% Perform alternating signed addition directly
    FOR(I = 1 : 1 : 3) {
        FOR(J = 1 : 1 : 3) {
            IF(I / 2 == 1) {
                S = S + (10 + (-5));  %% Add direct unsigned and signed values
            } ELSE {
                S = S + (-15 + 8);   %% Add direct signed and unsigned values
            }
        }
    }

    %% Write the sum
    WRITE("Sum: ", S);
}
